/**
 * 增加技能点（简单系统：直接提升技能等级）
 */
addSkillPoints(skillType, points) {
    if (!this.player.skills[skillType]) {
        this.player.skills[skillType] = 0;
    }
    
    this.player.skillPoints += points;
    this.player.skills[skillType] = Math.min(100, this.player.skills[skillType] + points);
    
    // 更新技能显示
    this.updateSkills();
}

/**
 * 获取技能名称
 */
getSkillName(skillType) {
    const names = {
        technical: '技术技能',
        management: '管理技能',
        sales: '销售技能',
        communication: '沟通技能'
    };
    return names[skillType] || skillType;
}

/**
 * 获取技能加成（每10点提供1%收入加成）
 */
getSkillBonus() {
    const totalSkill = 
        this.player.skills.technical +
        this.player.skills.management +
        this.player.skills.sales +
        this.player.skills.communication;
    
    return Math.floor(totalSkill / 10); // 每10点1%加成
}

/**
 * 更新技能显示
 */
updateSkills() {
    // 在工作面板中更新技能显示（如果存在）
    const skillsList = document.getElementById('skills-list');
    if (skillsList) {
        skillsList.innerHTML = '';
        
        Object.keys(this.player.skills).forEach(skillType => {
            const skillValue = this.player.skills[skillType];
            const item = document.createElement('div');
            item.style.cssText = 'display: flex; justify-content: space-between; padding: 5px 0; border-bottom: 1px solid #eee;';
            item.innerHTML = `
                <span>${this.getSkillName(skillType)}：</span>
                <span style="font-weight: bold; color: #667eea;">${skillValue}</span>
            `;
            skillsList.appendChild(item);
        });
    }
    
    // 更新技能点显示
    const skillPointsEl = document.getElementById('skill-points');
    if (skillPointsEl) {
        skillPointsEl.textContent = this.player.skillPoints;
    }
}

/**
 * 计算绩效奖金（每月根据工作表现获得）

calculatePerformanceBonus() {
    if (!this.player.currentJob) return 0;
    
    // 绩效奖金 = 工作表现 × 50元
    const bonus = Math.floor(this.player.workPerformance * 50);
    return bonus;
}

/**
 * 计算年终奖（工作满12个月获得）
 */
calculateYearEndBonus() {
    if (!this.player.currentJob) return 0;
    
    // 检查是否工作满12个月
    if (this.player.workMonths >= 12 && this.player.workMonths % 12 === 0) {
        // 年终奖 = 月薪 × (2-5倍) × 工作表现系数
        const baseSalary = this.player.currentJob.salary;
        const multiplier = 2 + (this.player.workPerformance / 100) * 3; // 2-5倍
        const bonus = Math.floor(baseSalary * multiplier);
        return bonus;
    }
    
    return 0;
}

/**
 * 计算副业收入（如果存在副业）
 */
calculateSideJobIncome() {
    if (!this.player.sideJob) return 0;
    return this.player.sideJob.income;
}

/**
 * 检查是否可以晋升
 */
checkPromotion() {
    if (!this.player.currentJob) return false;
    
    // 晋升条件：
    // 1. 工作年限 >= 12个月
    // 2. 工作表现 >= 70
    // 3. 技能等级 >= 30（任一技能）
    // 4. 当前职级 < 4（最高4级）
    
    if (this.player.workMonths < 12) return false;
    if (this.player.workPerformance < 70) return false;
    if (this.player.jobLevel >= 4) return false;
    
    // 检查是否有技能 >= 30
    const hasSkill = Object.values(this.player.skills).some(skill => skill >= 30);
    if (!hasSkill) return false;
    
    return true;
}

/**
 * 执行晋升
 */
promote() {
    if (!this.checkPromotion()) return false;
    
    this.player.jobLevel++;
    const oldSalary = this.player.currentJob.salary;
    this.player.currentJob.salary = Math.floor(oldSalary * 1.3); // 加薪30%
    
    // 技能点奖励
    const skillPoints = 10;
    this.player.skillPoints += skillPoints;
    
    // 随机提升一个技能
    const skillTypes = Object.keys(this.player.skills);
    const randomSkill = skillTypes[this.randomInt(0, skillTypes.length - 1)];
    this.addSkillPoints(randomSkill, skillPoints);
    
    alert(`恭喜晋升！职级提升，薪资从${oldSalary.toLocaleString()}元提升到${this.player.currentJob.salary.toLocaleString()}元，${this.getSkillName(randomSkill)}+${skillPoints}`);
    
    this.updatePlayerPanel();
    this.updateWorkPanel();
    return true;
}

/**
 * 显示跳槽面板
 */
showJobHuntPanel() {
    if (!this.data || !this.data.jobs) return;
    
    this.showPanel('jobhunt-panel');
    const availableJobs = document.getElementById('available-jobs');
    availableJobs.innerHTML = '';
    
    // 筛选符合条件的工作（薪资高于当前工作或技能要求）
    const currentSalary = this.player.currentJob.salary;
    const suitableJobs = this.data.jobs.filter(job => {
        // 必须符合学历要求
        if (this.player.schoolLevel < job.schoolRequirement) return false;
        
        // 薪资必须高于当前工作至少10%
        if (job.salary <= currentSalary * 1.1) return false;
        
        return true;
    });
    
    if (suitableJobs.length === 0) {
        availableJobs.innerHTML = '<p>目前没有适合跳槽的工作机会</p>';
        return;
    }
    
    suitableJobs.forEach(job => {
        const item = document.createElement('div');
        item.className = 'list-item';
        item.innerHTML = `
            <h4>${job.name}</h4>
            <p>薪资：${job.salary.toLocaleString()} 元/月</p>
            <p>要求：${this.getSchoolRequirementName(job.schoolRequirement)}</p>
            <p>薪资提升：${((job.salary / currentSalary - 1) * 100).toFixed(0)}%</p>
            <button class="btn btn-primary" data-job-name="${job.name}">跳槽</button>
        `;
        
        const jumpBtn = item.querySelector('button');
        jumpBtn.addEventListener('click', () => {
            this.jobHunt(job);
        });
        
        availableJobs.appendChild(item);
    });
}

/**
 * 执行跳槽
 */
jobHunt(newJob) {
    if (!this.player.currentJob) return;
    
    // 跳槽成功率检查（基于技能和属性）
    const totalSkill = Object.values(this.player.skills).reduce((sum, val) => sum + val, 0);
    const totalAttributes = Object.values(this.player.attributes).reduce((sum, val) => sum + val, 0);
    
    const skillBonus = totalSkill / 10; // 每10点技能+1%成功率
    const attrBonus = totalAttributes / 20; // 每20点属性+1%成功率
    const baseRate = 60; // 基础成功率60%
    
    const successRate = Math.min(95, baseRate + skillBonus + attrBonus);
    const passed = Math.random() * 100 < successRate;
    
    if (passed) {
        const oldSalary = this.player.currentJob.salary;
        this.player.currentJob = {
            name: newJob.name,
            salary: newJob.salary,
            schoolRequirement: newJob.schoolRequirement
        };
        
        // 重置工作月数
        this.player.workMonths = 0;
        
        alert(`跳槽成功！从${oldSalary.toLocaleString()}元/月跳槽到${newJob.salary.toLocaleString()}元/月，薪资提升${((newJob.salary / oldSalary - 1) * 100).toFixed(0)}%`);
        
        this.updatePlayerPanel();
        this.updateWorkPanel();
        this.showPanel('work-management');
    } else {
        alert(`跳槽失败！成功率${successRate.toFixed(0)}%，继续努力提升技能和属性吧！`);
        this.showPanel('work-management');
    }
}
